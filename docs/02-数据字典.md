## 2. 表结构 DDL（SQL Server）

```sql
-- ============ 系统 ============
CREATE TABLE sys_factory (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_code  NVARCHAR(64)  NOT NULL UNIQUE,   -- 工厂代码
  factory_name  NVARCHAR(128) NOT NULL,          -- 工厂名称
  license_tier  NVARCHAR(16)  NOT NULL DEFAULT 'trial',  -- 合同授权档：trial入门 / enterprise企业 / flagship旗舰（有效档见运行时/docs/89）
  license_expires_at_utc  DATETIME2 NULL,       -- 云端授权到期（UTC）；空=未激活
  last_license_ok_utc     DATETIME2 NULL,       -- 最近一次成功云校验的服务端 UTC
  last_license_seen_utc   DATETIME2 NULL,       -- 最近一次发起校验的本地 UTC（防拨回时钟）
  trial_expires_at_utc    DATETIME2 NULL,       -- 体验账套到期（UTC）；空=不进清理队列（docs/90）
  trial_cleaned_at_utc    DATETIME2 NULL,       -- 体验账套已清场时间（防重复清理）
  created_at    DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE sys_user (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id    BIGINT NOT NULL,                 -- 所属工厂
  account       NVARCHAR(64)  NOT NULL,          -- 账号(可填手机号)
  phone         NVARCHAR(20)  NULL,              -- 手机号
  wechat_id     NVARCHAR(64)  NULL,              -- 微信号（可空，仅记录；不做登录/推送）
  name          NVARCHAR(64)  NOT NULL,          -- 姓名
  role          TINYINT NOT NULL,                -- 角色: 1管理员 2生产人员 3班组长
  password      NVARCHAR(128) NOT NULL,          -- 加密密码
  status        TINYINT NOT NULL DEFAULT 1,      -- 1启用 0停用
  created_at    DATETIME2 DEFAULT SYSDATETIME(),
  CONSTRAINT uk_user_account UNIQUE (factory_id, account)
);

CREATE TABLE sys_department (
  id          BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id  BIGINT NOT NULL,
  code        NVARCHAR(64) NOT NULL,             -- 部门编码
  name        NVARCHAR(128) NOT NULL,            -- 部门名称
  created_at  DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE sys_department_user (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  department_id BIGINT NOT NULL,
  user_id       BIGINT NOT NULL,
  CONSTRAINT uk_dept_user UNIQUE (department_id, user_id)
);

-- ============ 基础数据 ============
CREATE TABLE base_unit (
  id          BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id  BIGINT NOT NULL,
  name        NVARCHAR(64) NOT NULL,             -- 单位名称, 如"只"
  created_at  DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE base_defect_item (
  id          BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id  BIGINT NOT NULL,
  name        NVARCHAR(128) NOT NULL,            -- 不良品项名称, 如"粘锅"
  created_at  DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE base_operation (
  id          BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id  BIGINT NOT NULL,
  code        NVARCHAR(64) NOT NULL,             -- 工序编号
  name        NVARCHAR(128) NOT NULL,            -- 工序名称
  created_at  DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE base_operation_department (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  operation_id  BIGINT NOT NULL,
  department_id BIGINT NOT NULL,
  CONSTRAINT uk_op_dept UNIQUE (operation_id, department_id)
);  -- 工序-报工权限部门(仅该部门人员可报工)

CREATE TABLE base_operation_defect (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  operation_id  BIGINT NOT NULL,
  defect_id     BIGINT NOT NULL,
  CONSTRAINT uk_op_defect UNIQUE (operation_id, defect_id)
);  -- 工序-不良品项关联

CREATE TABLE base_routing (
  id          BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id  BIGINT NOT NULL,
  code        NVARCHAR(64) NOT NULL,             -- 路线编号
  name        NVARCHAR(128) NOT NULL,            -- 路线名称
  created_at  DATETIME2 DEFAULT SYSDATETIME()
);

CREATE TABLE base_routing_step (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  routing_id    BIGINT NOT NULL,
  operation_id  BIGINT NOT NULL,
  seq           INT NOT NULL,                    -- 工序顺序(1开始)
  CONSTRAINT uk_routing_seq UNIQUE (routing_id, seq)
);  -- 工艺路线-工序明细(有序)

CREATE TABLE base_product (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id    BIGINT NOT NULL,
  code          NVARCHAR(64) NOT NULL,           -- 产品编号(唯一)
  name          NVARCHAR(128) NOT NULL,          -- 产品名称(可重复)
  unit_id       BIGINT NULL,                     -- 单位
  routing_id    BIGINT NULL,                     -- 工艺路线
  supplier      NVARCHAR(128) NULL,              -- 供应商(扩展属性)
  price         DECIMAL(12,2) NULL,              -- 单价(扩展属性)
  created_at    DATETIME2 DEFAULT SYSDATETIME(),
    CONSTRAINT uk_product_code UNIQUE (factory_id, code)
);

CREATE TABLE base_knowledge_file (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id    BIGINT NOT NULL,                 -- 工厂
  file_name     NVARCHAR(256) NOT NULL,          -- 显示名（可与磁盘文件名不同）
  relative_path NVARCHAR(512) NOT NULL,          -- 相对 KnowledgeBase:RootPath 的路径，不存绝对路径
  file_type     NVARCHAR(16) NOT NULL,           -- pdf / docx / xlsx / jpg / png / webp（图片仅供预览，不参与 AI 文字问答）
  ref_type      NVARCHAR(32) NOT NULL,           -- product / operation
  ref_id        BIGINT NOT NULL,                 -- 挂的产品/工序 ID
  version       NVARCHAR(32) NULL,               -- 版本号
  created_by    BIGINT NULL,                     -- 登记人
  created_at    DATETIME2 DEFAULT SYSDATETIME()
);
CREATE INDEX ix_knowledge_file_ref ON base_knowledge_file(ref_type, ref_id);
-- 口径：一个文件挂多处 = 插多条记录（relative_path 不唯一，ref_type+ref_id 不同）

-- ============ 工价规则（工资计价基础数据） ============
CREATE TABLE base_price_rule (
  id             BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id     BIGINT NOT NULL,
  product_id     BIGINT NULL,                   -- 产品；NULL=不限
  operation_id   BIGINT NULL,                   -- 工序；NULL=不限
  department_id  BIGINT NULL,                   -- 部门/班组；NULL=不限
  user_id        BIGINT NULL,                   -- 个人；NULL=不限
  price_type     TINYINT NOT NULL,              -- 计价方式: 1计件 2计时 3固定
  unit_price     DECIMAL(12,4) NOT NULL,        -- 单价: 计件=元/件, 计时=元/小时, 固定=单笔额
  deduct_price   DECIMAL(12,4) NULL,            -- 不良品扣款单价(元/件); NULL=不扣
  effective_from DATETIME2 NOT NULL,            -- 生效时间
  effective_to   DATETIME2 NULL,                -- 失效时间; NULL=当前有效
  priority       INT NOT NULL DEFAULT 0,        -- 匹配优先级(值小者优先; 具体度相同时用)
  created_at     DATETIME2 DEFAULT SYSDATETIME()
);
CREATE INDEX ix_price_rule_factory ON base_price_rule(factory_id, effective_from);
-- 匹配口径：product_id/operation_id/department_id/user_id 中 NULL 表示该维度不限；
--   报工时取"当前生效 + 命中维度最多(最具体)"的那条，具体度相同取 priority 小者。
-- 注意：base_product.price 是产品扩展属性，与工价无关；工价一律走本表。

-- ============ 生产业务 ============
CREATE TABLE prod_work_order (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id    BIGINT NOT NULL,
  order_no      NVARCHAR(64) NOT NULL,           -- 工单编号
  product_id    BIGINT NOT NULL,                 -- 产品
  qty           INT NOT NULL,                    -- 下单数量
  status        TINYINT NOT NULL DEFAULT 0,      -- 0未开始 1执行中 2已结束 3已取消
  due_date      DATETIME2 NULL,                  -- 计划交期；空=未设交期(不参与临期/超期判断)
  created_by    BIGINT NULL,                     -- 创建人
  created_at    DATETIME2 DEFAULT SYSDATETIME(),
  CONSTRAINT uk_order_no UNIQUE (factory_id, order_no)
);

CREATE TABLE prod_work_order_operation (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  work_order_id BIGINT NOT NULL,                 -- 工单
  operation_id  BIGINT NOT NULL,                 -- 工序
  seq           INT NOT NULL DEFAULT 1,          -- 工序顺序
  plan_qty      INT NOT NULL DEFAULT 0,          -- 该工序计划数
  assignee_user_id BIGINT NULL,                  -- 派工执行人冗余首个（docs/29/202）；空=未派工；真源见 prod_work_order_operation_assignee
  CONSTRAINT uk_wop_work_operation UNIQUE (work_order_id, operation_id),
  CONSTRAINT uk_wop_work_seq UNIQUE (work_order_id, seq)
);

-- 工单工序派工明细（一道工序可派多人，docs/202）
CREATE TABLE prod_work_order_operation_assignee (
  id                       BIGINT IDENTITY(1,1) NOT NULL CONSTRAINT PK_prod_wop_assignee PRIMARY KEY,
  work_order_operation_id  BIGINT NOT NULL,
  user_id                  BIGINT NOT NULL,
  CONSTRAINT UQ_prod_wop_assignee UNIQUE (work_order_operation_id, user_id)
);

CREATE TABLE prod_report (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id    BIGINT NOT NULL,
  order_id      BIGINT NOT NULL,                 -- 工单
  operation_id  BIGINT NOT NULL,                 -- 工序
  user_id       BIGINT NOT NULL,                 -- 报工人
  good_qty      INT NOT NULL DEFAULT 0,          -- 良品数
  defect_qty    INT NOT NULL DEFAULT 0,          -- 不良品数
  defect_id     BIGINT NULL,                     -- 不良品原因
  duration_minutes INT NOT NULL DEFAULT 0,       -- 报工时长（分钟）；0=未填
  batch_no      NVARCHAR(32) NULL,               -- 批量补报批次号；单条报工为空；同一次补报共用
  product_id    BIGINT NULL,                     -- 产品快照(冗余, 工资结算免join工单→产品)
  department_id BIGINT NULL,                     -- 报工部门快照(人会调岗, 工资按当时部门算)
  unit_price    DECIMAL(12,4) NULL,              -- 命中工价单价快照(计件=元/件, 计时=元/小时)
  wage_amount   DECIMAL(12,2) NULL,              -- 预计算工资额(结算回填; NULL=未计算)
  settled_flag  BIT NOT NULL DEFAULT 0,          -- 0未结算(可改) 1已结算(冻结, 禁改/禁删)
  review_status TINYINT NOT NULL DEFAULT 0,      -- 复核: 0待复核 1已通过 2已退回
  reviewed_by   BIGINT NULL,                     -- 复核人
  reviewed_at   DATETIME2 NULL,                  -- 复核时间
  reject_reason NVARCHAR(256) NULL,              -- 退回原因(仅 status=2)
  report_time   DATETIME2 DEFAULT SYSDATETIME(), -- 报工时间
  client_request_id NVARCHAR(64) NULL,          -- 客户端提交唯一标识(去重；旧端不传则跳过)
  share_batch_no NVARCHAR(32) NULL,              -- 多人分摊/代报同批号；与补报 batch_no 独立；单人自报空
  operator_user_id BIGINT NULL                  -- 代报/分摊操作人；单人自报空；业务归属仍看 user_id
);
CREATE INDEX ix_report_order ON prod_report(order_id);
CREATE INDEX ix_report_user_time ON prod_report(user_id, report_time);
CREATE INDEX ix_report_review ON prod_report(factory_id, review_status, report_time);
CREATE UNIQUE INDEX uk_report_client_req ON prod_report(factory_id, client_request_id, user_id) WHERE client_request_id IS NOT NULL;
CREATE INDEX ix_report_share_batch ON prod_report(share_batch_no) WHERE share_batch_no IS NOT NULL;

-- 报工修改日志（docs/205）：每次改数追加一条，不覆盖历史
CREATE TABLE prod_report_change_log (
  id                   BIGINT IDENTITY(1,1) NOT NULL CONSTRAINT PK_prod_report_change_log PRIMARY KEY,
  factory_id           BIGINT NOT NULL,
  report_id            BIGINT NOT NULL,                 -- 被改报工
  changed_by           BIGINT NOT NULL,                 -- 修改人
  old_good_qty         INT NOT NULL,
  old_defect_qty       INT NOT NULL,
  old_defect_id        BIGINT NULL,
  old_duration_minutes INT NOT NULL,
  new_good_qty         INT NOT NULL,
  new_defect_qty       INT NOT NULL,
  new_defect_id        BIGINT NULL,
  new_duration_minutes INT NOT NULL,
  source               TINYINT NOT NULL,                -- 来源: 1=PC 2=H5
  changed_at           DATETIME2 NOT NULL CONSTRAINT DF_prod_report_change_log_ca DEFAULT SYSDATETIME()
);
CREATE INDEX ix_report_change_log_report ON prod_report_change_log(report_id, changed_at DESC);
CREATE INDEX ix_report_change_log_factory ON prod_report_change_log(factory_id, changed_at DESC);

-- ============ 工资结算（salary_） ============
CREATE TABLE salary_statement (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id    BIGINT NOT NULL,
  user_id       BIGINT NOT NULL,                 -- 领薪人
  period_type   TINYINT NOT NULL,                -- 结算周期: 1月 2周
  period_value  NVARCHAR(16) NOT NULL,           -- 周期值: 月'2026-09' / 周'2026-W37'
  total_amount  DECIMAL(12,2) NOT NULL DEFAULT 0,-- 应发合计 = Σ(salary_statement_item.amount − deduct_amount)（docs/136）
  status        TINYINT NOT NULL DEFAULT 0,      -- 0草稿 1已确认 2已发放
  confirmed_at  DATETIME2 NULL,                  -- 确认时间
  created_at    DATETIME2 DEFAULT SYSDATETIME(),
  CONSTRAINT uk_statement_period UNIQUE (factory_id, user_id, period_type, period_value)
);
-- 注：原 5 手工列 base_salary/meal_allowance/other_allowance/social_tax/other_deduction
-- 已由 docs/136 移除。DbCompat：历史全 0 则 DROP；有非 0 则保留列停用（不映射实体）。

CREATE TABLE salary_statement_item (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  statement_id  BIGINT NOT NULL,                 -- 所属工资单
  report_id     BIGINT NOT NULL,                 -- 来源报工记录(全局唯一: 一条报工最多进一张工资单)
  rule_id       BIGINT NULL,                     -- 命中的工价规则ID(追溯用, 不设硬外键)
  calc_qty      INT NOT NULL DEFAULT 0,          -- 参与计算数量(计件=良品数; 计时=0; 固定=1)
  unit_price    DECIMAL(12,4) NOT NULL,          -- 结算单价快照
  amount        DECIMAL(12,2) NOT NULL DEFAULT 0,-- 小计(扣款前)
  deduct_amount DECIMAL(12,2) NULL,              -- 不良扣款小计
  CONSTRAINT uk_statement_item_report UNIQUE (report_id)  -- 一条报工只能结算一次
);
CREATE INDEX ix_statement_item_statement ON salary_statement_item(statement_id);

-- ============ 工人异常上报（docs/28） ============
CREATE TABLE prod_abnormal (
  id             BIGINT IDENTITY(1,1) NOT NULL CONSTRAINT PK_prod_abnormal PRIMARY KEY,
  factory_id     BIGINT NOT NULL,
  abnormal_type  TINYINT NOT NULL,              -- 1设备故障 2物料短缺 3质量异常
  description    NVARCHAR(500) NOT NULL,
  image_path     NVARCHAR(512) NULL,             -- 相对路径，如 uploads/abnormal/{factoryId}/xx.jpg
  work_order_id  BIGINT NULL,                   -- 可选关联工单
  reported_by    BIGINT NOT NULL,
  reported_at    DATETIME2 NOT NULL CONSTRAINT DF_prod_abnormal_ra DEFAULT SYSDATETIME(),
  status         TINYINT NOT NULL CONSTRAINT DF_prod_abnormal_st DEFAULT 0, -- 0待处理 1已恢复
  handled_by     BIGINT NULL,
  recovered_at   DATETIME2 NULL,
  handle_note    NVARCHAR(256) NULL
);
CREATE INDEX ix_abnormal_factory_status ON prod_abnormal(factory_id, status, reported_at DESC);

-- ============ 扩展能力(自定义字段) ============
CREATE TABLE sys_custom_field (
  id            BIGINT IDENTITY(1,1) PRIMARY KEY,
  factory_id    BIGINT NOT NULL,
  target        NVARCHAR(32) NOT NULL,           -- 归属对象: work_order/product/operation/report
  field_name    NVARCHAR(64) NOT NULL,           -- 字段名称（显示名，可改）
  field_type    NVARCHAR(16) NOT NULL,           -- 类型: single(单选)/text(文本)/number(数字)
  field_key     NVARCHAR(32) NULL,              -- 业务编码标识；空=普通字段。色码汇总约定：color=颜色、spec=规格/尺码（docs/102）
  show_in_order_list BIT NOT NULL CONSTRAINT DF_sys_custom_field_show_list DEFAULT 0, -- 仅 work_order 有意义；1=PC工单列表显示该列
  created_at    DATETIME2 DEFAULT SYSDATETIME()
);
-- 同厂同 target 下非空 field_key 唯一（过滤唯一索引，见 DbCompat）
-- CREATE UNIQUE INDEX uk_custom_field_key ON sys_custom_field(factory_id, target, field_key) WHERE field_key IS NOT NULL;

CREATE TABLE sys_custom_field_option (
  id          BIGINT IDENTITY(1,1) PRIMARY KEY,
  field_id    BIGINT NOT NULL,
  label       NVARCHAR(64) NOT NULL,             -- 选项值, 如"一号桌"
  seq         INT NOT NULL DEFAULT 0
);

CREATE TABLE sys_custom_field_value (
  id          BIGINT IDENTITY(1,1) PRIMARY KEY,
  field_id    BIGINT NOT NULL,
  target_id   BIGINT NOT NULL,                   -- 归属对象记录ID（report 字段时=报工记录 Id）
  value       NVARCHAR(256) NULL,                -- 字段值
  CONSTRAINT uk_field_target UNIQUE (field_id, target_id)
);

-- ============ 工单打印标签设置（工厂级一条） ============
CREATE TABLE sys_print_setting (
  id                BIGINT IDENTITY(1,1) NOT NULL CONSTRAINT PK_sys_print_setting PRIMARY KEY,
  factory_id        BIGINT NOT NULL,
  label_size        NVARCHAR(32) NOT NULL CONSTRAINT DF_sys_print_setting_size DEFAULT N'Label80x60', -- Label30x40 | Label80x60
  qr_mode           NVARCHAR(32) NOT NULL CONSTRAINT DF_sys_print_setting_qr DEFAULT N'OrderNo',     -- OrderNo | ReportUrl | ReportUrlOp
  base_url          NVARCHAR(256) NULL,                                                              -- ReportUrl / ReportUrlOp 时的前端基址
  show_product_code BIT NOT NULL CONSTRAINT DF_sys_print_setting_pc DEFAULT 1,
  show_product_name BIT NOT NULL CONSTRAINT DF_sys_print_setting_pn DEFAULT 1,
  show_qty          BIT NOT NULL CONSTRAINT DF_sys_print_setting_qty DEFAULT 1,
  show_ops          BIT NOT NULL CONSTRAINT DF_sys_print_setting_ops DEFAULT 1,
  show_custom_field_ids NVARCHAR(512) NULL,  -- JSON 数组，工单自定义字段 fieldId，如 [1,2]；空=不打自定义
  updated_at        DATETIME2 NOT NULL CONSTRAINT DF_sys_print_setting_ua DEFAULT SYSDATETIME(),
  CONSTRAINT UQ_sys_print_setting_factory UNIQUE (factory_id)
);

-- ============ 登录页设置（工厂级一条；左侧大图自配，docs/69） ============
CREATE TABLE login_setting (
  id          BIGINT IDENTITY(1,1) NOT NULL CONSTRAINT PK_login_setting PRIMARY KEY,
  factory_id  BIGINT NOT NULL,                 -- 工厂(账套)，隔离

## 4. 删除顺序（外键依赖倒序）

```
base_knowledge_file（知识库文件 → 引用产品/工序，先删）
salary_statement_item（工资明细 → 引用报工）

## 141 群通道字段补充

- `sys_wechat_alert_setting.to_user`：测试、交期默认群内@成员；规则不继承。`sys_wechat_alert_rule.to_user`：该规则@成员，空=仅群通知；group允许UserId或@all，多人归一为 `|`，@all出现时仅保留@all。
- `sys_wechat_send_attempt.robot_key_hash`：可空SHA-256哈希，仅标识已开始的群发送尝试。历史、人工核实和未外发记录可空；接口忽略此内部字段。
- `sys_wechat_send_attempt.created_at`：群外发记录为持久化发送开始时间，用于滚动一分钟计数；人工记录为处理时间。以hash与时间计数，独立于每日charged额度，成功、失败、未知及测试均计入分钟上限。
- 启动升级顺序129→131→141；旧session规则停用，未外发投递取消并记录原因，已进入发送阶段但未落结果的投递标unknown；成功/未知历史保留，不转换group，不删除旧应用字段。
